46def set_to_ones(Tcsr: csr_matrix, tol: float = 1e-8) -> None: 

47 """Replace near-one sparse-matrix entries in place. 

48 

49 Parameters 

50 ---------- 

51 Tcsr : :class:`scipy.sparse.csr_matrix` 

52 Sparse matrix whose stored values are modified in place. 

53 tol : float, default=1e-8 

54 Absolute tolerance around one. Stored values ``x`` satisfying 

55 ``abs(x - 1.0) <= tol`` are replaced by exactly ``1.0``. 

56 

57 Returns 

58 ------- 

59 None 

60 The input matrix is modified in place. 

61 """ 

62 Tcsr.data[np.abs(Tcsr.data - 1.0) <= tol] = 1.0

65def csc_row_normalize(X: csr_matrix | csc_matrix) -> csc_matrix: 

66 """Return a row-normalized sparse matrix in CSC format. 

67 

68 Parameters 

69 ---------- 

70 X : :class:`scipy.sparse.csr_matrix` or :class:`scipy.sparse.csc_matrix` 

71 Sparse matrix to normalize by row sums. Rows with zero sum are left 

72 unchanged. 

73 

74 Returns 

75 ------- 

76 :class:`scipy.sparse.csc_matrix` 

77 Row-normalized copy of ``X`` in CSC format. 

78 """ 

79 X_csr = X.tocsr() 

80 

81 n_rows = X_csr.shape[0] 

82 for i in range(n_rows): 

83 row_sum = X_csr.data[X_csr.indptr[i]:X_csr.indptr[i+1]].sum() 

84 if row_sum != 0: 

85 X_csr.data[X_csr.indptr[i]:X_csr.indptr[i+1]] /= row_sum 

86 

87 return X_csr.tocsc()

90def remove_nnz_rowcol( 

91 L: csr_matrix | csc_matrix, 

92) -> tuple[csr_matrix | csc_matrix, np.ndarray, int]: 

93 """Remove rows and columns that are both structurally zero. 

94 

95 A row and column index is retained when either the corresponding row or 

96 column contains at least one stored value. 

97 

98 Parameters 

99 ---------- 

100 L : :class:`scipy.sparse.csr_matrix` or :class:`scipy.sparse.csc_matrix` 

101 Square sparse matrix from which structurally zero rows and columns are 

102 removed. 

103 

104 Returns 

105 ------- 

106 L_small : :class:`scipy.sparse.csr_matrix` or \ 

107 :class:`scipy.sparse.csc_matrix` 

108 Sparse submatrix containing only retained rows and columns. 

109 nonzero_indices : :class:`numpy.ndarray` 

110 One-dimensional integer array with the retained row and column indices 

111 in the original matrix. 

112 size : int 

113 Original linear matrix size, equal to ``L.shape[0]``. 

114 """ 

115 # indices with zero sum row AND col 

116 nonzerosum_rowcols = ~np.logical_and(L.getnnz(1) == 0, 

117 L.getnnz(0) == 0) 

118 

119 nonzero_indices, = (nonzerosum_rowcols).nonzero() 

120 L_small = L[nonzerosum_rowcols][:, nonzerosum_rowcols] 

121 

122 return ( 

123 L_small, 

124 nonzero_indices, 

125 L.shape[0], 

126 )

129def set_to_zeroes( 

130 T: SparseStochMat | csr_matrix | csc_matrix, 

131 tol: float | None = 1e-8, 

132 relative: bool = True, 

133 use_absolute_value: bool = False, 

134) -> None: 

135 """Replace near-zero sparse-matrix entries in place. 

136 

137 When ``tol`` is not ``None``, stored values close to zero are replaced by 

138 exact zeros and eliminated from SciPy sparse matrices. For 

139 :class:`stochmat.SparseStochMat`, the operation is delegated to its own 

140 ``set_to_zeroes`` method. 

141 

142 Parameters 

143 ---------- 

144 T : :class:`stochmat.SparseStochMat`, \ 

145 :class:`scipy.sparse.csr_matrix`, or \ 

146 :class:`scipy.sparse.csc_matrix` 

147 Sparse matrix whose stored values are modified in place. 

148 tol : float or None, default=1e-8 

149 Threshold below which values are set to zero. If ``None``, the function 

150 returns without modifying ``T``. 

151 relative : bool, default=True 

152 If ``True``, scale ``tol`` by the largest absolute stored value before 

153 thresholding SciPy sparse matrices. 

154 use_absolute_value : bool, default=False 

155 If ``True``, threshold values by ``abs(value) <= tol``. If ``False``, 

156 threshold values by ``value <= tol``. 

157 

158 Returns 

159 ------- 

160 None 

161 The input matrix is modified in place when ``tol`` is not ``None``. 

162 

163 Raises 

164 ------ 

165 TypeError 

166 If ``T`` is not a supported sparse matrix type. 

167 """ 

168 if tol is not None: 

169 threshold = float(tol) 

170 if isinstance(T, SparseStochMat): 

171 T.set_to_zeroes(threshold, relative=relative) 

172 elif isinstance(T, (csr_matrix, csc_matrix)): 

173 data = T.data 

174 if data.size > 0: 

175 if relative: 

176 # tol = tol*np.abs(T.data).max() 

177 # finding the max of the absolute value without making a 

178 # copy of the whole array 

179 threshold = ( 

180 threshold 

181 * np.abs([data.min(), data.max()]).max() 

182 ) 

183 

184 if use_absolute_value: 

185 data[np.abs(data) <= threshold] = 0 

186 else: 

187 data[data <= threshold] = 0 

188 

189 T.eliminate_zeros() 

190 else: 

191 raise TypeError("T must be csc,csr or SparseStochMat")

194def to_dense(M: ArrayLike | spmatrix | sparray | SparseStochMat) -> np.ndarray: 

195 """Coerce a sparse or dense matrix-like object to an ndarray. 

196 

197 Parameters 

198 ---------- 

199 M : array-like, :class:`scipy.sparse.spmatrix`, \ 

200 :class:`scipy.sparse.sparray`, or \ 

201 :class:`stochmat.SparseStochMat` 

202 Matrix-like object to convert. NumPy-compatible array-like inputs are 

203 passed to :func:`numpy.asarray`; SciPy sparse matrices and arrays are 

204 converted with ``toarray``; and :class:`stochmat.SparseStochMat` 

205 objects are converted with ``to_full_mat``. 

206 

207 Returns 

208 ------- 

209 :class:`numpy.ndarray` 

210 Dense array representation of ``M``. Existing NumPy arrays are returned 

211 unchanged. 

212 """ 

213 if isinstance(M, np.ndarray): 

214 return M 

215 if isinstance(M, SparseStochMat): 

216 M = M.to_full_mat() 

217 if isinstance(M, (spmatrix, sparray)): 

218 return M.toarray() 

219 return np.asarray(M)

222def find_spectral_gap(L: csr_matrix | csc_matrix) -> np.ndarray: 

223 """Compute the spectral gap of a connected random-walk Laplacian. 

224 

225 Parameters 

226 ---------- 

227 L : :class:`scipy.sparse.csr_matrix` or :class:`scipy.sparse.csc_matrix` 

228 Connected random-walk Laplacian matrix. The matrix is converted to CSR 

229 format internally. 

230 

231 Returns 

232 ------- 

233 :class:`numpy.ndarray` 

234 One-element array returned by :func:`scipy.sparse.linalg.eigsh`, 

235 containing the eigenvalue nearest zero of the adjusted symmetric 

236 Laplacian. 

237 """ 

238 Lcsr = L.tocsr() 

239 n_nodes = L.shape[0] 

240 

241 identity = eye(n_nodes, dtype=np.float64, format="csr") 

242 

243 degs = np.diff((identity - Lcsr).indptr) 

244 

245 D12 = diags(np.sqrt(degs), 

246 format="csr") 

247 Dm12 = diags(1/np.sqrt(degs), 

248 format="csr") 

249 

250 Lsym = D12 @ Lcsr @ Dm12 

251 

252 # stationary solution 

253 Pi = np.vstack([degs/degs.sum()]*n_nodes) 

254 

255 gap = eigsh(Lsym.toarray()-Pi, 1, sigma=0, return_eigenvectors=False) 

256 

257 return gap
